31.c　求数组最大值

C
#include <stdio.h>



//求最值

//需求：已知数组元素为{33,5,22,66,55}

//请找出数组中最大值并打印在控制台



int main(){



    int arr[]={33,5,22,66,55};

    int max = arr[0];

    int len = sizeof(arr)/sizeof(int);



    for(int i=0;i<len;i++){

        if(max<arr[i+1]){

            max = arr[i+1];

        }



    }

    printf("%d\n",max);

    return 0;

}

易错点 注意 越界隐患：循环 i<len 内访问 arr[i+1]，当 i=len-1 时读了 arr[len]（越界，未定义行为）。因为 max 初值取了 arr[0]，应把条件改成 i<len-1。多数时候碰巧不出错，但这是真实的错题点。

32.c　随机数入数组并求和

C
#include <stdio.h>

#include <time.h>

#include <stdlib.h>



//遍历数组求和（基础版）

//需求：生成10个1~100之间的随机数存入数组

//1）求出所有数据的和



int main(){



    int arr[10]={0};

    int len = sizeof(arr)/sizeof(int);



    srand(time(NULL));



    for(int i=0;i<len;i++){

        int num=rand()%100+1;

        arr[i]=num;

    }



    int sum=0;

    for(int i=0;i<len;i++){

        sum += arr[i];

    }



    printf("%d\n",sum);







    return 0;

}

33.c　随机数组统计（和 / 平均 / 计数）

C
#include <stdio.h>

#include <stdlib.h>

#include <time.h>



// 遍历数组求和（升级版）

// 需求：生成10个1~100之间的随机数存入数组，要求数据不能重复

// 1）求出所有数据的和

// 2）求所有数据的平均数

// 3）统计有多少个数据比平均值小



int main(){



    int arr[10]={0};

    int len = sizeof(arr)/sizeof(int);



    srand(time(NULL));



    int sum=0;

    int j=0;



    for(int i=0;i<len;i++){

        arr[i]=rand()%100+1;

        sum+=arr[i];

        printf("%d ",arr[i]);

    }

    int ave=sum/len;

    for(int i=0;i<len;i++){

        if(arr[i]<ave){

            j++;

        }

    }



    printf("\n");

    printf("随机数的和为：%d\n",sum);

    printf("平均数为：%d\n",ave);

    printf("有%d个数比平均值小\n",j);







    return 0;

}

易错点 注意 注释需求写明「数据不能重复」，但代码没有去重逻辑——生成时应查重重试（或用标记数组）。这是一个需求与实现不符的典型错例。

34.c　键盘录入 + 数组反转

C
#include <stdio.h>



// 需求：键盘录入5个数据并存入数组，完成以下需求

// 1）遍历数组

// 2）反转数组

// 3）再次遍历



void printarr(int arr[] , int len){

    for(int i=0;i<len;i++){

        printf("%d\n",arr[i]);

    }

}



int main(){



    int arr[5]={0};

    int len = sizeof(arr)/sizeof(int);



    for(int i=0;i<len;i++){

        printf("请录入第%d个元素：",i+1);

        scanf("%d",&arr[i]);

    }



    printarr(arr,len);



    int i=0;

    int j=len-1;

    while(i<j){

        int temp=arr[i];

        arr[i]=arr[j];

        arr[j]=temp;

        i++;

        j--;

    }



    printarr(arr,len);



    return 0;

}

辨析延伸 要点 双下标对撞交换：i 从头、j 从尾，while(i<j) 交换后相向而行；还封装了 printarr(int arr[], int len) 复用打印。

35.c　打乱数组（Fisher-Yates 思路）

C
#include <stdio.h>

#include <stdlib.h>

#include <time.h>



//打乱数组内的数据



int main(){



    int arr[]={1,2,3,4,5};

    int len=sizeof(arr)/sizeof(int);

    srand(time(NULL));



    for(int i=0;i<len;i++){

        int index = rand()%len;

        int temp = arr[i];

        arr[i]=arr[index];

        arr[index]=temp;

    }



    for(int i=0;i<len;i++){

        printf("%d\n",arr[i]);

    }



    return 0;

}

辨析延伸 要点 每个位置 i 与随机下标 index 交换。严格来说 rand()%(len-i)+i 才是均匀洗牌，当前写法有轻微偏差——作为练习足够。

36.c　基本查找（顺序查找）

C
#include <stdio.h>



//基本查找



int order(int arr[],int len,int num){

    for(int i=0;i<len;i++){

        if(arr[i]==num){

            return i;

        }

    }

    return -1;

}



int main(){



    int arr[]={41,5,7454,14,4};

    int len = sizeof(arr)/sizeof(int);

    int num=5;

    int index=order(arr,len,num);

    printf("%d\n",index);



    return 0;

}

辨析延伸 要点 找到返回下标，找不到返回 -1——用 -1 做哨兵值是查找函数的标准约定。

37.c　二分查找

C
#include <stdio.h>



int con(int arr[],int len,int num){

    int max = len-1;

    int min = 0;

    int mid = (max + min)/2;

    for(;min<=max;){

        if(arr[mid]<num){

            min=mid+1;

            mid=(min+max)/2;

        }

        else if(arr[mid]>num){

            max=mid-1;

            mid=(min+max)/2;

        }

        else{

            return mid;

            break;

        }

    }

    return -1;



}

int main(){

    int arr[]={12,34,56,78,90,123,456,789};

    int len=sizeof(arr)/sizeof(int);

    int num=123;

    int index=con(arr,len,num);

    printf("%d\n",index);



    return 0;

}

辨析延伸 要点 仅适用于有序数组；min/max/mid 三指针收缩区间。return mid; 后面的 break; 永远执行不到，可删。

38.c　冒泡排序

C
#include <stdio.h>



// 冒泡排序：

//  1，相邻的元素两两比较，大的放右边，小的放左边。

//  2，第一轮比较完毕之后，最大值就已经确定，第二轮可以少循环一次，后面以此类推。

//  3，如果数组中有n个数据，总共我们只要执行n-1轮的代码就可以。

// 需求：

//  利用冒泡排序将下列数据按照从小到大的顺序进行排序

//      3,5,2,1,4



int main(){



    int arr[]={3,5,2,1,4};

    int len = sizeof(arr)/sizeof(int);



    for(int j=1;j<len;j++){

        for(int i=0;i<len-j;i++){

            if(arr[i]>arr[i+1]){

                int temp=arr[i];

                arr[i]=arr[i+1];

                arr[i+1]=temp;

            }

        }

    }

    for(int i=0;i<len;i++){

        printf("%d ",arr[i]);

    }



    return 0;

}

辨析延伸 要点 相邻两两比较、大的沉底；外层 n-1 轮，内层每轮少比一次（i<len-j）。

39.c　选择排序

C
#include <stdio.h>



// 选择排序：

    // 1，从0索引开始，跟后面的元素——比较。

    // 2，小的放前面，大的放后面

    // 3，第一轮循环从0索引开始比较，结束后最小的数据已经确定。

    // 4，第二轮循环从1索引开始，后面以此类推。

    // 5，第三轮循环从2索引开始，后面以此类推。

    // 6，第四轮循环从3索引开始，后面以此类推。

// 需求：

    // 利用选择排序将下列数据按照从小到大的顺序进行排序

    // 3,5,2,1,4



int main(){



    int arr[]={3,5,2,1,4};

    int len = sizeof(arr)/sizeof(int);



    for(int j=0;j<len-1;j++){

        for(int i=j+1;i<len;i++){

            if(arr[j]>arr[i]){

                int temp=arr[j];

                arr[j]=arr[i];

                arr[i]=temp;

            }

        }

    }

    for(int i=0;i<len;i++){

        printf("%d ",arr[i]);

    }



    return 0;

}

辨析延伸 要点 每轮用基准位 j 与后面所有位比较，小的换到前面——与冒泡的区别：它和「固定基准」比，不是和「相邻」比。
