HM 01　基础语法与输入输出

1.c　输出保留 3 位小数

C
#include <stdio.h>



int main(){

    printf("%.3f\n",1.9);

    return 0;

}

辨析延伸 要点 %.3f 是 printf 精度控制：小数点后保留 3 位，四舍五入。

1_1.c　二 / 八 / 十六进制字面量

C
#include <stdio.h>

int main(){

    int a = 0x10; //16进制

    int b = 010; //8进制

    int c = 0b10; //2进制

    printf("%d %d %d\n",a,b,c);

    return 0;

}

辨析延伸 要点 0x10=16、010=8、0b10=2（C23 前二进制是编译器扩展），用 %d 都按十进制输出。

2.c　输入年龄并回显

C
#include <stdio.h>



int main(){

    int age;

    printf("请输入您的年龄：");

    scanf("%d",&age);

    printf("您的年龄为：%d\n",age);

    return 0;

}

辨析延伸 要点 scanf("%d", &a) 必须写 & 取地址；提示语用 printf 单独输出。

3.c　读入两个整数求和

C
#include <stdio.h>



int main(){

    int a,b = 0;

    scanf("%d %d",&a,&b);

    printf("%d\n",a+b);

    return 0;



}

辨析延伸 要点 int a, b = 0; 只有 b 初始化了，a 未初始化——但随后 scanf 会覆盖，无碍。规范的写法是两个都初始化。

4.c　长方体各面面积与体积

C
#include <stdio.h>



int main(){

    float a,b,c;

    printf("请输入长&宽&高(保留2位小数)：");

    scanf("%f %f %f",&a,&b,&c);

    float s1=a*b;

    float s2=b*c;

    float s3=a*c;

    float v=a*b*c;

    printf("A的面积为%.2f\nB的面积为%.2f\nC的面积为%.2f\n体积为%.2f\n",s1,s2,s3,v);

    return 0;

}



//用double更好！

辨析延伸 要点 原注释就是最好的总结：「用 double 更好」——float 精度只有约 7 位有效数字。

5.c　三位数拆个位 / 十位 / 百位

C
#include <stdio.h>



//键盘录入一个三位数，将其拆分为个位、十位、百位并打印。



int main(){

    int num = 0;

    printf("请输入一个三位数：");

    scanf("%d",&num);

    int a = num%10;

    int b = (num%100)/10;

    int c = num/100;

    printf("个位为：%d\n十位为：%d\n百位为：%d\n",a,b,c);

    return 0;

}

辨析延伸 要点 拆位三板斧：个位 n%10、十位 n/10%10、百位 n/100。

6.c　字符的 ASCII 码值

C
#include <stdio.h>



int main(){

    printf("%d\n",'a');

    printf("%d\n",'A');

    printf("%d\n",'0');

    return 0;

}

辨析延伸 要点 字符本质是整数：'a'=97、'A'=65、'0'=48，大小写相差 32。

7.c　逻辑表达式：两位数是否与 7 有关

C
#include <stdio.h>



int main(){

    printf("请输入一个两位整数：");

    int a = 0;

    scanf("%d",&a);

    printf("%d\n",(a/10) != (7) && (a%10) != (7));

    printf("%d\n",((a/10) != (7) && (a%10) != (7)) && (a%7)!= 0 );

    return 0;

}

辨析延伸 要点 「不含 7」= (a/10)!=7 && (a%10)!=7；「既不含 7 又不是 7 的倍数」再 && (a%7)!=0。逻辑运算的结果是 0 或 1。

8.c　三元运算符嵌套求最大值

C
#include <stdio.h>



int main(){

    int a = 0;

    int b = 0;

    int c = 0;

    scanf("%d %d %d",&a,&b,&c);



    //超纲了

    // if (a>b && a>c)

    // {

    //     printf("%d\n",a);



    // }

    // else if(b>a && b>c)

    // {

    //     printf("%d\n",b);

    // }

    // else{

    //     printf("%d\n",c);

    // }



    //三元运算符！！！

    int t = a > b ? a : b ;

    int s = t > c ? t : c ;

    printf("%d\n",s);



    return 0;

}

辨析延伸 要点 int t = a>b ? a : b; 再拿 t 和 c 比，两步搞定三数取大，比 if-else 链简洁。
