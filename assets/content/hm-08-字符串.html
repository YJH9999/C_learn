HM 08　字符串

52.c　字符数组 vs 指针字符串

C
#include <stdio.h>



int main(void)

{

    // 1.利用字符数组 + 双引号的方式定义字符串

    char str1[4] = "abc";

    printf("修改前 str1：%s\n", str1);



    // 字符数组内容可以修改

    str1[0] = 'Q';

    printf("修改后 str1：%s\n\n", str1);





    // 2.利用指针 + 双引号的方式定义字符串

    char* str2 = "abcd";

    char* str3 = "abcd";



    printf("str2 的地址：%p\n", str2);

    printf("str3 的地址：%p\n", str3);

    // str2 和 str3 地址一样，常量区复用

    // str2[0] = 'Q';   // 错误！只读常量区，不能修改，打开会崩溃



    return 0;

}

重难点 注意 重点辨析：char str1[4]="abc" 在栈上、可修改；char* str2="abcd" 指向只读常量区，修改会崩溃。相同字面量编译器只存一份（str2 与 str3 地址相同）。
