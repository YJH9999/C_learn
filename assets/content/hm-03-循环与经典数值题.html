16.c　判断一个数是否为 2 的幂

C
#include <stdio.h>​int main(){    //2的幂​    int a = 0;​    printf("Please input a number:");​    scanf("%d",&a);​    if(a==1){        printf("yes!");    }​    else{        while (a%2==0 && a > 1)​        {            a=a/2;        }​        if(a==1){            printf("yes");        }​        else{            printf("no");        }    }​    return 0;}

辨析延伸 要点 不断除 2 降幂，最后剩下 1 就是 2 的幂。进阶写法：n>0 && (n & (n-1))==0。

17.c　折纸过珠穆朗玛峰

C
#include <stdio.h>​//珠穆朗玛峰-折纸次数​int main(){    int a = 1;​    int i = 0;​    while(a<88444300){        a *= 2;​        i++;    }​    printf("%d\n",i);​    return 0;}

辨析延伸 要点 纸张厚度 1（单位可约去），每折一次 ×2，while 模拟到 8844430 为止——指数增长的威力。

18!!!.c　⭐ 整数反转 123 → 321

C
#include <stdio.h>​//123->321​int main(){    int a = 0;​    printf("Please input a number among all the range:");​    scanf("%d",&a);​    // int b = a/100;//百位​    // int c = a/10%10;//十位​    // int d = a%10;//个位​    // printf("%d%d%d\n",d,c,b);​    int c = 0;​    while(a != 0){        int b = a%10;​        a = a/10;​        c = c*10+b;    }​    printf("%d\n",c);​    return 0;}

辨析延伸 要点 通用套路（不限三位数）：b = a%10 取末位 → c = c*10 + b 拼到结果 → a /= 10 去掉末位，循环直到 a 为 0。注释里还保留了仅限三位数的拆位解法。

19.c　不用 sqrt 的整数开方

C
#include <stdio.h>​int main(){    //开方，去掉小数部分​    int a = 0;​    printf("请输入一个非负整数：");​    scanf("%d",&a);​    int b = 0;​    while(b*b<=a){        b++;    }​    printf("%d\n",b-1);​    return 0;}

辨析延伸 要点 b*b <= a 一直试探，退出时 b-1 就是整数部分；输入很大时 b*b 可能溢出 int，可改用 long long。

24.c　统计区间内质数个数

C
#include <stdio.h>​//质数个数​int main(){    //输入最大值​    int num = 0;​    printf("请输入一个正整数：");​    scanf("%d",&num);​    int a = 0;//定义质数个数​    for(int i=2;i<=num;i++){//进入1-最大值范围的循环​        int count = 0;//定义内循环中对被除数取余为0的除数的个数​        for(int j=2;j<i;j++){//进入1-i范围的循环​            if(i%j==0){                count++;​                break;//只要count!=0，就跳出该循环            }        }​        if(count==0){//一次内循环中若除去1和i本身，无其他的公约数​            a++;        }    }​    printf("在1-%d的范围内共有%d个质数",num,a);​    return 0;}

辨析延伸 要点 双重循环：外层枚举 2~n，内层用 2~i-1 试除，遇到因子就 break（count!=0 即合数）。这是试除法质数判定的入门模板。

25.c　幂之和 1¹ + 2² + … + 10¹⁰

C
#include <stdio.h>​//1**1+2**2+...+10**10​int main(){    long long a=0;​    for(int i=1;i<=10;i++){        long long b=1;​        for(int j=1;j<=i;j++){            b*=i;        }​        a+=b;    }​    printf("%lld",a);​    return 0;}

辨析延伸 要点 10¹⁰ 已超出 int（约 21 亿）甚至 long 的范围，必须用 long long 配 %lld 输出——不换类型就是溢出错题。
