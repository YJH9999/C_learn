#include <stdio.h>



int main(){

    int arr[500];

    int j = 0;

    for(int i = 2000; i<=2500;i++){

        printf("年份%d\n",i);

        if(i%4!=0){

            printf("不是闰年\n");

        }

        else if(i%4==0&&i%100!=0){

            printf("是闰年\n");

            arr[j]=i;

            j++;

        }

        else if(i%4==0&&i%100==0){

            printf("是闰年\n");

            arr[j]=i;

            j++;

        }

        else{

            printf("不是闰年\n");

        }

    }

    for(int k = 0; k < j; k++){

        printf("%d ", arr[k]);

    }

    return 0;

}
