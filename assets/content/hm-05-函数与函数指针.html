27.c　用函数比较两个长方形面积

C
#include <stdio.h>



void square(int length1,int width1,int length2,int width2){

    int square1 = length1*width1;

    int square2 = length2*width2;

    if(square1<square2){

        printf("第二个大");

    }

    else if(square1>square2){

        printf("第一个大");

    }

    else{

        printf("一样大");

    }



}



int main(){



    int a=0;

    int b=0;

    printf("请输入第一个长方形的长和宽：");

    scanf("%d %d",&a,&b);

    int c=0;

    int d=0;

    printf("请输入第二个长方形的长和宽：");

    scanf("%d %d",&c,&d);



    square(a,b,c,d);



    return 0;

}

辨析延伸 要点 把「比较」封装成无返回值、printf 直接输出的函数——注意这种设计把输出耦合进了函数，更通用的做法是返回 -1/0/1。

50.c　函数指针基础

C
#include<stdio.h>



void method1();

int method2(int num1, int num2);



int main()

{

    // 1.定义函数指针指向两个函数

    void (*p1)() = method1;

    int (*p2)(int, int) = method2;



    // 2.利用函数指针去调用函数

    p1();

    int num = p2(10,20);

    printf("%d\n", num);



    return 0;

}



void method1()

{

    printf("method1\n");

}



int method2(int num1, int num2)

{

    return num1 + num2;

}

辨析延伸 要点 void (*p1)() = method1; 声明函数指针：返回值 + (* 名字) + (参数表)；之后 p1() 等价于 method1()。

51.c　函数指针数组实现计算器

C
// 定义加、减、乘、除、四个函数

// 用户键盘录入三个数字

// 前两个表示参与计算的数字

// 第三个数字表示调用的函数

// 1: 加法

// 2: 减法

// 3: 乘法

// 4: 除法

#include <stdio.h>



int add(int a,int b){

    return a+b;

}

int subtract(int a,int b){

    return a-b;

}

int multiply(int a,int b){

    return a*b;

}

int divide(int a,int b){

    return a/b;

}



int main(){

    //函数指针数组

    int(*p[4])(int,int)={add,subtract,multiply,divide};

    int a,b,c;

    scanf("%d %d %d",&a,&b,&c);

    int res=(p[c-1])(a,b);

    printf("%d",res);

    return 0;

}

辨析延伸 要点 int (*p[4])(int,int) = {add, subtract, multiply, divide};——函数指针数组按下标调度函数，switch 都省了；调用写 (p[c-1])(a, b)。
