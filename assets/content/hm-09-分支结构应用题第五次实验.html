1.c　Game Damage Calculator（游戏伤害计算器）

基础伤害 = 攻击力 × 0.5；boss ×1.5；技能 ×2；向下取整输出。

C
#include <stdio.h>

int main() {
    int atk, is_boss, use_skill;
    float damage;

    scanf("%d %d %d", &atk, &is_boss, &use_skill);

    damage = atk * 0.5;             /* 基础伤害 */
    if (is_boss == 1) damage *= 1.5;  /* boss 增伤 50% */
    if (use_skill == 1) damage *= 2;  /* 技能翻倍 */

    printf("%d\n", (int)damage);      /* 向下取整输出 */
    return 0;
}

2.c　Package Delivery Fee（快递运费计算）

基础费 10 元；超重每 kg +5（不足按 1kg）；超距每 5km +3；偏远 +20。ceil 向上取整。

C
#include <stdio.h>
#include <math.h>   /* for ceil() */

int main() {
    float weight, distance;
    int remote, fee = 10;

    scanf("%f %f %d", &weight, &distance, &remote);

    if (weight > 1)  fee += ceil(weight - 1) * 5;
    if (distance > 5) fee += ceil((distance - 5) / 5) * 3;
    if (remote == 1) fee += 20;

    printf("%d\n", fee);
    return 0;
}

3.c　Pizza Order（披萨订单）

S/M/L = 30/45/60 元；配料每个 +5（最多按 3 个）；周二 -10；保底不为负。

C
#include <stdio.h>

int main() {
    char size;
    int toppings, day, price = 0;

    scanf(" %c %d %d", &size, &toppings, &day);  /* 空格跳过空白 */

    if (size == 'S') price = 30;
    else if (size == 'M') price = 45;
    else if (size == 'L') price = 60;

    if (toppings > 3) toppings = 3;
    price += toppings * 5;
    if (day == 2) price -= 10;
    if (price < 0) price = 0;

    printf("%d\n", price);
    return 0;
}

4.c　Traffic Light Timer（交通灯倒计时）

红 60s / 绿 45s / 黄 5s；输入灯状态和已亮时间，超时长输出 invalid。

C
#include <stdio.h>

int main() {
    int light, time_passed;
    scanf("%d %d", &light, &time_passed);

    if (light == 1) {                       /* 红灯 */
        if (time_passed < 0 || time_passed > 60) printf("invalid\n");
        else printf("%d\n", 60 - time_passed);
    }
    else if (light == 2) {                  /* 绿灯 */
        if (time_passed < 0 || time_passed > 45) printf("invalid\n");
        else printf("%d\n", 45 - time_passed);
    }
    else if (light == 3) {                  /* 黄灯 */
        if (time_passed < 0 || time_passed > 5) printf("invalid\n");
        else printf("%d\n", 5 - time_passed);
    }
    else printf("invalid\n");

    return 0;
}

5.c　Vending Machine（自动售货机）

switch 定价 + if 判余额：可乐 5 / 薯片 8 / 巧克力 10 / 水 3，输出找零或余额不足。

C
#include <stdio.h>

int main() {
    int item, money, price;
    scanf("%d %d", &item, &money);

    switch (item) {
        case 1: price = 5;  break;
        case 2: price = 8;  break;
        case 3: price = 10; break;
        case 4: price = 3;  break;
        default: printf("Invalid item!\n"); return 0;
    }

    if (money >= price) printf("Success, change: %d\n", money - price);
    else printf("Insufficient money\n");

    return 0;
}

6.c　Movie Rating（电影分级）

按 G / PG / R 输出分级说明；PG 需要再读一个字符二次判断。

C
#include <stdio.h>

int main() {
    char rating;
    scanf(" %c", &rating);

    switch (rating) {
        case 'G':
            printf("General Audiences - All ages admitted.\n");
            break;
        case 'P':                       /* 先匹配 'P' 再判断后续 */
            if (getchar() == 'G')
                printf("Parental Guidance Suggested.\n");
            else
                printf("invalid\n");
            break;
        case 'R':
            printf("Restricted - Under 17 requires accompanying adult.\n");
            break;
        default:
            printf("invalid\n");
    }
    return 0;
}

7.c　Weather Alert（天气预警系统）

switch 天气代码 + if 温度阈值：晴 >35℃ / 雨 <10℃ / 雪 <-5℃ 高温、寒雨、暴雪预警。

C
#include <stdio.h>

int main() {
    int weather_code;
    float temp;
    scanf("%d %f", &weather_code, &temp);

    switch (weather_code) {
        case 1:
            printf("Sunny day. ");
            if (temp > 35) printf("High temperature warning! Stay indoors.\n");
            else printf("Good weather for outdoor activities.\n");
            break;
        case 2:
            printf("Rainy day. ");
            if (temp < 10) printf("Cold rain warning! Wear warm clothes.\n");
            else printf("Don't forget your umbrella.\n");
            break;
        case 3:
            printf("Snowy day. ");
            if (temp < -5) printf("Blizzard warning! Avoid traveling.\n");
            else printf("Roads may be slippery.\n");
            break;
        case 4: printf("Windy day. Strong wind alert!\n"); break;
        case 5: printf("Hazy day. Haze warning! Wear a mask.\n"); break;
        default: printf("Invalid weather code!\n");
    }
    return 0;
}

8.c　Adventure Game Choice（冒险游戏选择）

switch 四方向剧情分支，部分分支用 rand() 做随机事件判定。

C
#include <stdio.h>
#include <stdlib.h>
#include <time.h>

int main() {
    int choice;
    srand(time(NULL));
    scanf("%d", &choice);

    switch (choice) {
        case 1:
            printf("You found a treasure box! ");
            printf("Got %d gold coins!\n", 10 + rand() % 41);
            break;
        case 2:
            printf("An NPC appears! ");
            if (rand() % 2 == 0) printf("You received a key.\n");
            else printf("You received a map.\n");
            break;
        case 3:
            printf("A monster attacks! ");
            if (rand() % 2 == 0) printf("You won the battle!\n");
            else printf("You lost the battle...\n");
            break;
        case 4:
            printf("You returned to the starting point.\n");
            break;
        default:
            printf("Invalid direction!\n");
    }
    return 0;
}

9.c　Fast Food Order（快餐点餐系统）

switch 嵌套：套餐 1 汉堡再选饮料大小（S/M/L +3/5/7），套餐 2 可加酱，套餐 3 固定价。

C
#include <stdio.h>

int main() {
    int package, add_sauce, price = 0;
    char size;

    scanf("%d", &package);

    switch (package) {
        case 1:                          /* 汉堡套餐 */
            price = 30;
            scanf(" %c", &size);
            switch (size) {
                case 'S': price += 3; break;
                case 'M': price += 5; break;
                case 'L': price += 7; break;
                default: printf("Invalid size!\n"); return 0;
            }
            printf("%d\n", price);
            break;
        case 2:                          /* 炸鸡套餐 */
            price = 40;
            scanf("%d", &add_sauce);
            if (add_sauce == 1) price += 2;
            printf("%d\n", price);
            break;
        case 3:                          /* 披萨套餐 */
            printf("50 (free drink included)\n");
            break;
        default:
            printf("Invalid package!\n");
    }
    return 0;
}
