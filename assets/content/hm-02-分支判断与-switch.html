HM 02　分支判断与 switch

10.c　双人电影票相邻判断

C
#include <stdio.h>​​​int main(){​    int p1 = 0;​    int z1 = 0;​    printf("请输入您的电影票位置（第几排及座位号）：");​    scanf("%d %d",&p1,&z1);​    int p2 = 0;​    int z2 = 0;​    printf("请输入您女朋友的电影票位置（第几排及座位号）：");​    scanf("%d %d",&p2,&z2);​    if (p1==p2&&(z1==z2+1||z1==z2-1)){​        printf("看电影");​    }​    else{​        printf("打游戏");​    }​​​    return 0;​}

辨析延伸 要点 同一排且座位号相差 1：p1==p2 && (z1==z2+1 || z1==z2-1)——注意 && 优先级高于 ||，但加括号更清晰。

11.c　成绩分级嵌套 if-else

C
#include <stdio.h>​​​int main(){​    int score = 0;​    printf("请输入成绩：");​    scanf("%d",&score);​​​    if(score>=0&&score<=100){​        if(score >=95 && score <= 100){​            printf("Bicycle");​        }​        else if(score >=90 && score <= 94){​            printf("Visit a funny park");​        }​        else if(score >=85 && score <= 89){​            printf("Changeable Robot");​        }​        else if(score >=80 && score <= 85){​            printf("None");​        }        ​        else{​            printf("Bitten");​        }​    }​    else{​        printf("Wrong data");​    }​​​    return 0;​}

易错点 注意 区间重叠：score>=80 && score<=85 与 score>=85 && score<=89 在 85 分重叠；由于 else-if 顺序判断，85 实际落入 85~89 分支，80~85 分支里的 85 永远不可达。区间边界建议一半用闭、一半用开。

12.c　switch 判断星期（基础）

C
#include <stdio.h>​​​int main(){​​​    int week = 0;​    printf("Please input the data of week:");​    scanf("%d",&week);​​​    switch(week){​        case 1:​            printf("111");​            break;​        case 2:​            printf("222");​            break;​        case 3:​            printf("333");​            break;​        case 4:​            printf("444");​            break;​        case 5:​            printf("555");​            break;​        case 6:​            printf("666");​            break;​        case 7:​            printf("777");​            break;​        default:​            printf("Not Found");​            break;​    }​​​    return 0;​}

13.c　switch 判断运动项目

C
#include <stdio.h>​​​int main(){​​​    int week = 0;​    printf("Please input the data of week:");​    scanf("%d",&week);​​​    switch(week){​        case 1:​            printf("Running");​            break;​        case 2:​            printf("Swimming");​            break;​        case 3:​            printf("Walking");​            break;​        case 4:​            printf("Riding");​            break;​        case 5:​            printf("Boxing");​            break;​        case 6:​            printf("Climbing");​            break;​        case 7:​            printf("Eating");​            break;​        default:​            printf("False");​            break;​    }​​​    return 0;​}

14.c　case 穿透判断四季

C
#include <stdio.h>​​​int main(){​​​int month = 0;​printf("Please input the month:");​scanf("%d",&month);​​​switch(month){​    case 1:​    case 2:​    case 12:​        printf("Winter\n");​        break;​    case 3:​    case 4:​    case 5:​        printf("Spring\n");​        break;​    case 6:​    case 7:​    case 8:​        printf("Summer\n");​        break;​    case 9:​    case 10:​    case 11:​        printf("autumn\n");​        break;​    default:​        printf("Not Found\n");​        break;​}​​​    return 0;​}

辨析延伸 要点 case 1 / 2 / 12 共用一段代码——故意省略 break 让语句穿透，这是 case 穿透的标准用法；别忘最后加 default。
