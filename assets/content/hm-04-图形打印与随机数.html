22.c　双重循环打印星形图案

C
#include <stdio.h>​​​int main(){​​​    //打印：​    //*****​    //*****​    //*****​    //一次打印一个*​    // for(int i=1;i<=3;i++){​    //     for(int j=1;j<=5;j++){​    //         printf("*");​    //     }​    //     printf("\n");​    // }​​​    //打印:​    //*****​    //****​    //***​    //**​    //*​    for(int i=5;i>=1;i--){​        for(int j=1;j<=i;j++){​            printf("*");​        }​        printf("\n");​    }​​​    return 0;​}

辨析延伸 要点 外层控制行、内层控制列；注释里是 3×5 的矩形，主体是倒直角三角形——内层上限随外层变化是图形题核心。

23.c　九九乘法表

C
#include <stdio.h>​​​//九九乘法表​​​int main(){​​​    for(int i=1;i<=9;i++){​        for(int j=1;j<=i;j++){​            printf("%d*%d=%d\t",j,i,i*j);​        }​        printf("\n");​    }​​​    return 0;​}

辨析延伸 要点 printf("%d*%d=%d\t", j, i, i*j)，内层 j<=i 打出下三角；\t 让每列对齐。

28.c　生成指定范围 / 个数的随机数

C
#include <stdio.h>​#include <stdlib.h>​#include <time.h>​​​​​int main(){​​​    //0.获取范围​    int x=0;​    int y=0;​    int z=0;​    printf("请输入最小值：");​    scanf("%d",&x);​    printf("请输入最大值：");​    scanf("%d",&y);​    printf("请输入想获取随机数的个数：");​    scanf("%d",&z);​​​    //1.设置种子​    srand(time(NULL));​​​    for(int i=1;i<=z;i++){​        //2.获取随机数​        int num=rand()%(y-x+1)+x;​​​        //3.打印​        printf("%d\n",num);​    }​​​​​    return 0;​}

辨析延伸 要点 rand()%(y-x+1)+x 映射到 [x, y]；srand(time(NULL)) 用当前时间做种子，避免每次运行结果相同。

29.c　猜数字游戏

C
#include <stdio.h>​#include <stdlib.h>​#include <time.h>​​​//生成1-100的随机数，键盘录入猜数字​​​int main(){​​​    srand(time(NULL));​    int num=rand()%(45)+15;​​​    for(int i=1;;i++){​        printf("请输入您猜测的数字：");​        int a=0;​        scanf("%d",&a);​​​        if(a>num){​            printf("猜大了\n");​        }​        else if(a<num){​            printf("猜小了\n");​        }​        else{​            printf("猜中了！\n");​            printf("共猜了%d次\n",i);​            break;​        }​    }​​​​​​​    return 0;​}

辨析延伸 要点 for(int i=1;;i++) 省略条件的无限循环 + 猜中 break——循环计数器还能顺便统计次数。
